#include <WiFi.h>
#include <WebServer.h>

// 에이스봇 Car Shield 전용 74HC595 시프트 레지스터 제어 핀 정의
const int DATA_PIN = 5;
const int STCP_PIN = 17;
const int SHCP_PIN = 18;
const int EN_PIN   = 16;

// 노트북 핫스팟 정보
const char* ssid     = "DESKTOP-H8FAL5H 6854";
const char* password = "0%68Ws16";

WebServer server(80);

// 시프트 레지스터 칩에 명령 데이터를 밀어 넣는 에이스봇 전용 함수
void updateMotor(int motor_data) {
  digitalWrite(STCP_PIN, LOW);
  shiftOut(DATA_PIN, SHCP_PIN, MSBFIRST, motor_data);
  digitalWrite(STCP_PIN, HIGH);
}

// 🛠️ 주행 핸들러 (전진/후진 유지, 좌우 제자리 회전 비트값만 보정)
void handleForward() { updateMotor(165); server.send(200, "text/plain", "forward"); }
void handleBackward() { updateMotor(90); server.send(200, "text/plain", "backward"); }
void handleLeft() { updateMotor(106); server.send(200, "text/plain", "left"); }       // ⭐ 102 -> 106으로 보정 (왼쪽 바퀴 구동 비트 추가)
void handleRight() { updateMotor(149); server.send(200, "text/plain", "right"); }     // ⭐ 153 -> 149로 보정 (왼쪽 바퀴 구동 비트 추가)
void handleStop() { updateMotor(0); server.send(200, "text/plain", "stop"); }

void setup() {
  Serial.begin(115200);
  
  pinMode(SHCP_PIN, OUTPUT);
  pinMode(EN_PIN, OUTPUT);
  pinMode(DATA_PIN, OUTPUT);
  pinMode(STCP_PIN, OUTPUT);
  
  digitalWrite(EN_PIN, LOW); 
  updateMotor(0);            
  
  WiFi.begin(ssid, password);
  Serial.print("Connecting to Hotspot");
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("\nWi-Fi Connected!");
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP()); 

  server.on("/", handleRoot);
  server.on("/forward", handleForward);
  server.on("/backward", handleBackward);
  server.on("/left", handleLeft);
  server.on("/right", handleRight);
  server.on("/stop", handleStop);
  
  server.begin();
}

void loop() {
  server.handleClient();
}
